## Solution

```cpp

class Task {
public:
    int freq;
    int time;

    Task(int f, int t) {
        freq = f;
        time = t;
    }
};

class Solution {
public:
    int leastInterval(vector<char>& tasks, int n) {
        int time = 0;

        priority_queue<Task, vector<Task>, function<bool(Task, Task)>> maxHeap(
            [](Task a, Task b) {
                return a.freq < b.freq;
            }
        );

        queue<Task> q;

        int counts[26] = {0};

        for (char t : tasks) {
            counts[t - 'A']++;
        }

        for (int c : counts) {
            if (c > 0) {
                maxHeap.push(Task(c, 1));
            }
        }

        while (!maxHeap.empty() || !q.empty()) {
            time++;

            while (!q.empty() && q.front().time == time) {
                maxHeap.push(q.front());
                q.pop();
            }

            if (!maxHeap.empty()) {
                Task task = maxHeap.top();
                maxHeap.pop();

                task.freq--;
                task.time = time + n + 1;

                if (task.freq > 0) {
                    q.push(task);
                }
            }
        }

        return time;
    }
};
```
