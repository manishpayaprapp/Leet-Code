## Solution

```cpp
class Solution {
public:
    bool carPooling(vector<vector<int>>& trips, int capacity) {
        map<int, int> mp;

        for (auto& trip : trips) {
            int pass = trip[0];
            int start = trip[1];
            int end = trip[2];

            mp[start] += pass;
            mp[end] -= pass;
        }

        int passengers = 0;

        for (auto& entry : mp) {
            passengers += entry.second;

            if (passengers > capacity) {
                return false;
            }
        }

        return true;
    }
};
```
