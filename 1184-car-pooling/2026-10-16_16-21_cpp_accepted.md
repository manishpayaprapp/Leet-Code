## Solution

```cpp
class Solution {
public:
    bool carPooling(vector<vector<int>>& trips, int capacity) {
        vector<int> buckets(1001);

        for (auto& trip : trips) {
            int pass = trip[0];
            int start = trip[1];
            int end = trip[2];

            buckets[start] += pass;
            buckets[end] -= pass;
        }

        int passengers = 0;

        for (auto& value : buckets) {
            passengers += value;

            if (passengers > capacity) {
                return false;
            }
        }

        return true;
    }
};
```
