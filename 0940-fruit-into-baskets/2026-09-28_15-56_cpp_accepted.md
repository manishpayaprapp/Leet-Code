## Solution

```cpp
class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        int maxx = 0;

        unordered_map<int, int> mp;

        int i = 0;
        int j = 0;


        while (j < fruits.size()) {

            // Add current fruit
            mp[fruits[j]]++;

            // More than 2 types of fruits
            while (mp.size() > 2) {

                mp[fruits[i]]--;

                if (mp[fruits[i]] == 0) {
                    mp.erase(fruits[i]);
                }

                i++;
            }

            // Current valid window = [i ... j]
            maxx = max(maxx, j - i + 1);

            j++;
        }

        return maxx;
    }
};
```
