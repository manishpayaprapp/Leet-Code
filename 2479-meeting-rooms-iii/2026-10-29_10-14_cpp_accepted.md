```cpp
class Solution {
public:
    int mostBooked(int n, vector<vector<int>>& meetings) {
        sort(meetings.begin(),meetings.end());

        vector<long> endTimesofRooms(n);
        vector<int> count(n);

        for(int i = 0; i < meetings.size(); i++){

            int startTime = meetings[i][0];
            int endTime = meetings[i][1];

            int minEndTimeRoomIdx = 0;
            bool isRoomAllocated = false;

            for(int room = 0; room < n; room++){

                if(startTime >= endTimesofRooms[room]){
                    endTimesofRooms[room] = endTime;
                    count[room] ++;
                    isRoomAllocated = true;
                    break;
                }

                if(endTimesofRooms[room] < endTimesofRooms[minEndTimeRoomIdx]){
                    minEndTimeRoomIdx = room;
                }
            }
            if(!isRoomAllocated){
                endTimesofRooms[minEndTimeRoomIdx] += endTime - startTime;
                count[minEndTimeRoomIdx] ++;
            }
        }
        int maxRoomCountIdx = 0;
        
        for(int i = 0; i < n; i++){
            if(count[i] > count[maxRoomCountIdx] ){
                maxRoomCountIdx = i;
            }
        }
        return maxRoomCountIdx;
    }
};
```
