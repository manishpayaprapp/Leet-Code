class Solution {
public:
    int mostBooked(int n, vector<vector<int>>& meetings) {
        sort(meetings.begin(), meetings.end());

        // Min-heap of available rooms (smallest room index first)
        priority_queue<int, vector<int>, greater<int>> availableRooms;

        // Min-heap of occupied rooms: {endTime, roomIndex}
        // Smallest end time first; if equal, smallest room index first
        priority_queue<
            pair<long long, int>,
            vector<pair<long long, int>>,
            greater<pair<long long, int>>
        > occupiedRooms;

        // Initially, all rooms are available
        for (int i = 0; i < n; i++) {
            availableRooms.push(i);
        }

        vector<int> counts(n, 0);

        for (int i = 0; i < meetings.size(); i++) {
            long long startTime = meetings[i][0];
            long long endTime = meetings[i][1];

            // Release all rooms whose meetings have ended
            while (!occupiedRooms.empty() &&
                   occupiedRooms.top().first <= startTime) {
                int room = occupiedRooms.top().second;
                occupiedRooms.pop();

                availableRooms.push(room);
            }

            if (!availableRooms.empty()) {
                // Allocate the smallest available room
                int room = availableRooms.top();
                availableRooms.pop();

                counts[room]++;

                occupiedRooms.push({endTime, room});
            } else {
                // No room is available.
                // Delay the meeting until the earliest room becomes free.
                auto [earliestEndTime, room] = occupiedRooms.top();
                occupiedRooms.pop();

                long long duration = endTime - startTime;
                long long newEndTime = earliestEndTime + duration;

                counts[room]++;

                occupiedRooms.push({newEndTime, room});
            }
        }

        // Find the room with the maximum number of meetings.
        // If tied, return the room with the smallest index.
        int maxRoomCountsIdx = 0;

        for (int i = 1; i < n; i++) {
            if (counts[i] > counts[maxRoomCountsIdx]) {
                maxRoomCountsIdx = i;
            }
        }

        return maxRoomCountsIdx;
    }
};
