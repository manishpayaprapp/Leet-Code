## Solution

```cpp

class MyCalendarThree {
    map<int, int> mp;

public:
    MyCalendarThree() {}

    int book(int startTime, int endTime) {
        mp[startTime]++;
        mp[endTime]--;

        int bookings = 0;
        int maxBookings = 0;

        for (auto entry : mp) {
            bookings += entry.second;
            maxBookings = max(maxBookings, bookings);
        }

        return maxBookings;
    }
};
```
