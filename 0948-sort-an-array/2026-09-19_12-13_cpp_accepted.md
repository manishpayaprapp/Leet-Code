## Solution

```cpp
class Solution {
public:
    vector<int> sortArray(vector<int>& nums) {
       sort(nums.begin(), nums.end());
       int n = nums.size();
       vector<int> arr(n);
       for(int i = 0; i< n;i++){
        arr[i] = nums[i];
       }
       return arr;
    }
};
```
