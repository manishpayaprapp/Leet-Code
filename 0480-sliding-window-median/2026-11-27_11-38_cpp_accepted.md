class Solution {
public:
    vector<double> medianSlidingWindow(vector<int>& nums, int k) {

        priority_queue<int> maxHeap;

        priority_queue<int, vector<int>, greater<int>> minHeap;

        // Stores elements pending deletion
        unordered_map<int, int> mp;

        int n = nums.size();

        // Build initial window
        for (int i = 0; i < k; i++) {
            int num = nums[i];

            if (maxHeap.empty()) {
                maxHeap.push(num);
                continue;
            }

            if (maxHeap.size() == minHeap.size()) {

                if (num > maxHeap.top()) {
                    minHeap.push(num);
                    maxHeap.push(minHeap.top());
                    minHeap.pop();
                } else {
                    maxHeap.push(num);
                }

            } else {

                maxHeap.push(num);
                minHeap.push(maxHeap.top());
                maxHeap.pop();
            }
        }

        vector<double> res(n - k + 1);

        int j = 0;

        // Initial median
        res[j] = (k % 2 == 1)
                    ? maxHeap.top()
                    : ((double)maxHeap.top() + minHeap.top()) / 2.0;

        // Sliding window
        for (int i = k; i < n; i++) {

            int numA = nums[i];
            int numR = nums[i - k];

            // Mark outgoing element for lazy deletion
            mp[numR]++;

            int counter = 0;

            // Insert incoming element
            if (numA <= maxHeap.top()) {
                maxHeap.push(numA);
                counter++;
            } else {
                minHeap.push(numA);
                counter--;
            }

            // Remove outgoing element logically
            if (numR <= maxHeap.top()) {
                counter--;
            } else {
                counter++;
            }

            // Balance heaps
            if (counter > 0) {
                minHeap.push(maxHeap.top());
                maxHeap.pop();
            }

            if (counter < 0) {
                maxHeap.push(minHeap.top());
                minHeap.pop();
            }

            // Remove stale elements from maxHeap
            while (!maxHeap.empty() &&
                   mp[maxHeap.top()] > 0) {

                int ele = maxHeap.top();
                maxHeap.pop();

                mp[ele]--;
            }

            // Remove stale elements from minHeap
            while (!minHeap.empty() &&
                   mp[minHeap.top()] > 0) {

                int ele = minHeap.top();
                minHeap.pop();

                mp[ele]--;
            }

            j++;

            // Calculate median
            res[j] = (k % 2 == 1)
                        ? maxHeap.top()
                        : ((double)maxHeap.top() + minHeap.top()) / 2.0;
        }

        return res;
    }
};
