## Solution

```cpp
class Solution {
public:
    int characterReplacement(string s, int k) {
        int mx = 0;

        int i = 0;
        int j = 0;

        int maxC = 0;

        vector<int> arr(26);
        while(j < s.size()){
            char c = s[j];

                arr[c-'A']++;
                maxC = max(maxC,arr[c-'A']);


            while((j - i + 1) - maxC > k){
                arr[s[i] - 'A'] --;
                i++;  
            }
            mx = max(mx,j-i+1);
            j++;
        }
        return mx;
    }
};
```
