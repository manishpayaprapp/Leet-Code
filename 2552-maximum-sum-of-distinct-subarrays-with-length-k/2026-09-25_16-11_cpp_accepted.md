## Solution

```cpp
class Solution {
public:
    long long maximumSubarraySum(std::vector<int>& nums, int k) {
        long long sum = 0;
        long long max_sum = 0;
        unordered_map<int, int> map;
        int dups = 0;
        
        // Process the first window of size k
        for (int i = 0; i < k; i = i + 1) {
            // Equivalent to map.put(key, map.getOrDefault(key, 0) + 1)
            map[nums[i]] = map[nums[i]] + 1;
            sum = sum + nums[i];
            
            if (map[nums[i]] > 1) {
                dups = dups + 1;
            }
        }
        
        if (dups == 0) {
            max_sum = std::max(max_sum, sum);
        }
        
        // Slide the window across the rest of the vector
        for (int i = k; i < nums.size(); i = i + 1) {
            int numToAdd = nums[i];
            int numToRemove = nums[i - k];
            
            // 1. Add the incoming element (Right side of window)
            map[numToAdd] = map[numToAdd] + 1;
            if (map[numToAdd] > 1) {
                dups = dups + 1;
            }
            sum = sum + numToAdd;
            
            // 2. Remove the outgoing element (Left side of window)
            if (map[numToRemove] > 1) {
                dups = dups - 1;
            }
            map[numToRemove] = map[numToRemove] - 1;
            sum = sum - numToRemove;
            
            // 3. Update maximum if window contains distinct elements
            if (dups == 0) {
                max_sum = std::max(sum, max_sum);
            }
        }
        
        return max_sum;
    }
};
```
