## Solution

```cpp
class MyCalendar {
private:
    vector<pair<int, int>> bookings;

public:
    MyCalendar() {}

    bool book(int startTime, int endTime) {
        for (auto &p : bookings) {
            if (startTime < p.second && endTime > p.first) {
                return false;
            }
        }
        bookings.push_back({startTime, endTime});

        return true;
    }
};
```
