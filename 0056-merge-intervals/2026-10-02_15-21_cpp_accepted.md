## Solution

```cpp
class Solution {
public:
    bool isOverlap(vector<int>& int1, vector<int>& int2) {
        return int1[1] >= int2[0] &&
               int2[1] >= int1[0];
    }

public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        bool isMergeHappened = true;

        while (isMergeHappened) {
            isMergeHappened = false;

            for (int i = 0; i < intervals.size(); i++) {
                int j = i + 1;

                while (j < intervals.size()) {

                    if (isOverlap(intervals[i], intervals[j])) {

                        intervals[i][0] = min(intervals[i][0],
                                              intervals[j][0]);

                        intervals[i][1] = max(intervals[i][1],
                                              intervals[j][1]);

                        intervals.erase(intervals.begin() + j);

                        isMergeHappened = true;
                    }
                    else {
                        j++;
                    }
                }
            }
        }

        return intervals;
    }
};
```
