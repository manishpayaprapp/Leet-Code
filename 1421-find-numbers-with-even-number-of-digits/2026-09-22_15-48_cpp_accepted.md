## Solution

```cpp
class Solution {
public:
    int findNumbers(vector<int>& nums) {
        int total = 0;

        for (int i = 0; i < nums.size(); i++) {
            int num = nums[i];
            int count = 0;

            while (num != 0) {
                num /= 10;
                count++;
            }

            if (count % 2 == 0) {
                total++;
            }
        }

        return total;
    }
};
```
