## Solution

```cpp
class MedianFinder {
    // Min heap
    // Max heap
    priority_queue<int, vector<int>, greater<int>> minHeap;
    priority_queue<int> maxHeap;

public:
    // T: O(1), S: O(1)
    MedianFinder() {
    }

    // T: O(log n), S: O(1) auxiliary
    void addNum(int num) {
        if (maxHeap.empty()) {
            maxHeap.push(num);
            return;
        }

        if (maxHeap.size() == minHeap.size()) {
            if (num > maxHeap.top()) {
                minHeap.push(num);
                maxHeap.push(minHeap.top());
                minHeap.pop();
            } else {
                maxHeap.push(num);
            }
        } else {
            maxHeap.push(num);
            minHeap.push(maxHeap.top());
            maxHeap.pop();
        }
    }

    double findMedian() {
        if (maxHeap.size() == minHeap.size()) {
            return (static_cast<double>(maxHeap.top()) + minHeap.top()) / 2.0;
        }

        return maxHeap.top();
    }
};

```
