## Solution

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    vector<ListNode*> splitListToParts(ListNode* head, int k) {
        ListNode* curr = head;
        int len = 0;

        while(curr != nullptr){
            len++;
            curr = curr -> next;
        }
        int uniformDistrubitionNumber = len / k;
        int remainingNumber = len % k;

        vector<ListNode*> res(k);
        curr = head;

        for(int i = 0; i < k; i++){
            int count = uniformDistrubitionNumber;

            if(remainingNumber > 0){
                count++;
                remainingNumber--;
            }

            ListNode* temp = curr;
            ListNode* prev = nullptr;
            int counter = 0;

            while(counter < count){
                prev = curr;
                curr = curr -> next;
                counter++;
            }

            if(prev != nullptr){
                prev -> next = nullptr;
            }

            res[i] = temp;
        }
        return res;
    }
};
```
