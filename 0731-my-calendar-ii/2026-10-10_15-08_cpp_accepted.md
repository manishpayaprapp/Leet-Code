## Solution

```cpp
class MyCalendarTwo {
private:
    vector<pair<int, int>> bookings;
    vector<pair<int, int>> overlapBookings;

public:
    MyCalendarTwo() {}

    bool book(int startTime, int endTime) {
        // Reject if the new booking overlaps an existing double booking
        for (auto &p : overlapBookings) {
            if (startTime < p.second && endTime > p.first) {
                return false;
            }
        }

        // Find intersections with existing bookings
        for (auto &p : bookings) {
            if (startTime < p.second && endTime > p.first) {
                overlapBookings.push_back({
                    max(startTime, p.first),
                    min(endTime, p.second)
                });
            }
        }

        // Add the new booking
        bookings.push_back({startTime, endTime});

        return true;
    }
};
```
