## Solution

```cpp
class MyCalendarTwo {
    map<int, int> mp;

public:
    MyCalendarTwo() {
    }

    bool book(int startTime, int endTime) {
        mp[startTime]++;
        mp[endTime]--;

        int bookings = 0;

        for (auto entry : mp) {
            bookings += entry.second;

            if (bookings > 2) {
                // Undo the current booking
                mp[startTime]--;
                mp[endTime]++;

                // Remove entries whose values become zero
                if (mp[startTime] == 0) {
                    mp.erase(startTime);
                }

                if (mp.find(endTime) != mp.end() && mp[endTime] == 0) {
                    mp.erase(endTime);
                }

                return false;
            }
        }

        return true;
    }
};
```
