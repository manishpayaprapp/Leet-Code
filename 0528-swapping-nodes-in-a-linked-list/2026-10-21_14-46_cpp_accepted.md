## Solution

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* swapNodes(ListNode* head, int k) {
        int len = 0;

        ListNode* curr = head;

        while(curr != nullptr){
            len++;
            curr = curr -> next;
        }

        ListNode* n1 = head;
        ListNode* n2 = head;
        
        for(int i = 1; i < k;i++){
            n1 = n1 -> next;
        }

        for(int i = 1;i < len - k + 1;i++){
            n2 = n2 -> next;
        }
        
        int temp = n1 -> val;
        n1 -> val = n2 -> val;
        n2 -> val = temp;

        return head;
    }
};
```
