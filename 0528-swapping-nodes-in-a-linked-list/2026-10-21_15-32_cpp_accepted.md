## Solution

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* swapNodes(ListNode* head, int k) {
        ListNode* n2 = head;
        int jumps = 1;
        while(jumps < k){
            jumps ++;
            n2 = n2 -> next;
        }

        ListNode* n1 = head;
        ListNode* curr = n2;

        while(curr -> next != nullptr){
            curr = curr -> next;
            n1 = n1 -> next;
        }
        int temp = n1 -> val;
        n1 -> val = n2 -> val;
        n2 -> val = temp;

        return head;
    }
};
```
