## Solution

```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int maxx = 0;
        unordered_map<int,int> map;
        int j = 0;
        int i = 0;

        while(j < s.size()){
            char c = s[j];

            if(map.find(c) != map.end()){
                if(map[c] >= i){
                    i = map[c] + 1;
                }
            }
            
            map[c] = j;
            maxx = max(maxx,j - i + 1);
            j++;
        }
        return maxx;
    }
};
```
