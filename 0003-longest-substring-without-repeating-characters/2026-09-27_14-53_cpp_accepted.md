## Solution

```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int maxx = 0;
        unordered_set<int> set;
        int j = 0;
        int i = 0;

        while(j < s.size()){
            char c = s[j];

            while(set.find(c) != set.end()){
                set.erase(s[i]);
                i++;
            }
            set.insert(c);
            maxx = max(maxx,j - i + 1);
            j++;
        }
        return maxx;
    }
};
```
