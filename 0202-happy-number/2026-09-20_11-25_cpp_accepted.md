## Solution

```cpp
class Solution {
public:
    int sumOfSquareOfDigits(int n){
        int sum = 0;
        while(n > 0){
            int digit = n % 10;
            sum += (digit * digit);
            n = n / 10;
        }
        return sum;
    }
    bool isHappy(int n) {
        int slow = n;
        int fast = n;

        while(fast != 1){
            slow = sumOfSquareOfDigits(slow);
            fast = sumOfSquareOfDigits(sumOfSquareOfDigits(fast));

            if(fast == 1){
                return true;
            }
            if(fast == slow){
                return false;
            }
        }
        return true;
    }
};
```
