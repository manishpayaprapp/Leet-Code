## Solution

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */

class Solution {
public:
    ListNode* reverseEvenLengthGroups(ListNode* head) {
        int len = 0;
        int probGroupSize = 1;

        ListNode* curr = head;

        while (curr != nullptr) {
            len++;
            curr = curr->next;
        }

        curr = head;
        ListNode* p = nullptr;

        // T: O(n), S: O(1)
        while (curr != nullptr) {
            int actualGroupSize = min(probGroupSize, len);

            if (actualGroupSize % 2 == 0) {
                // Reverse the group
                ListNode* prev = nullptr;
                int counter = 0;
                ListNode* temp1 = curr;

                while (counter < actualGroupSize) {
                    counter++;

                    ListNode* next = curr->next;
                    curr->next = prev;
                    prev = curr;
                    curr = next;
                }

                p->next = prev;
                temp1->next = curr;
                p = temp1;
            } else {
                // Skip the group
                int counter = 0;

                while (counter < actualGroupSize) {
                    counter++;
                    p = curr;
                    curr = curr->next;
                }
            }

            probGroupSize++;
            len -= actualGroupSize;
        }

        return head;
    }
};
```
