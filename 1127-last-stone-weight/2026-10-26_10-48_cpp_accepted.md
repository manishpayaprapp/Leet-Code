## Solution

```cpp
class Solution {
public:
    int lastStoneWeight(vector<int>& stones) {
        while (stones.size() > 1) {
            sort(stones.begin(), stones.end());

            int last = stones.back();
            stones.pop_back();

            int lastSecond = stones.back();
            stones.pop_back();

            if (last != lastSecond) {
                stones.push_back(last - lastSecond);
            }
        }

        return stones.empty() ? 0 : stones[0];
    }
};
```
