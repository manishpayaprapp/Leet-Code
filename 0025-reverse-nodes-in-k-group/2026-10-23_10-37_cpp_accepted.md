## Solution

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverseKGroup(ListNode* head, int k) {
        ListNode* curr = head;
        int len = 0;
        while(curr != nullptr){
            len ++;
            curr = curr -> next;
        }

        int times = len / k;

        curr = head;

        ListNode* dummy = new ListNode(0);
        dummy -> next = head;

        ListNode* p1 = dummy;

        for(int i = 0; i < times; i++){\
        
        ListNode* prev = dummy;
        ListNode* p2 = curr;

            int count = 0;
            while(count < k && curr != nullptr){
                count++;

                ListNode* next = curr -> next;
                curr -> next = prev;
                prev = curr;
                curr = next;
            }
            p1 -> next = prev;
            p2 -> next = curr;
            p1 = p2;
        }
        return dummy -> next;
    }
};
```
