## Solution

```cpp
class Solution {
public:
    vector<string> findRepeatedDnaSequences(string s) {
        int K = 10;

        if (s.length() <= K) {
            return {};
        }

        unordered_set<int> seen;
        unordered_set<string> result;

        unordered_map<char, int> mp = {
            {'A', 0},
            {'C', 1},
            {'G', 2},
            {'T', 3}
        };

        int rep = 0;

        // 4^9
        int highestPower = 1;
        for (int i = 1; i < K; i++) {
            highestPower *= 4;
        }

        // Encode first 10 characters
        for (int i = 0; i < K; i++) {
            rep = rep * 4 + mp[s[i]];
        }

        seen.insert(rep);

        // Sliding window
        for (int i = K; i < s.length(); i++) {

            // Remove leftmost character
            rep -= mp[s[i - K]] * highestPower;

            // Shift left
            rep *= 4;

            // Add new character
            rep += mp[s[i]];

            if (seen.find(rep) != seen.end()) {
                result.insert(s.substr(i - K + 1, K));
            }

            seen.insert(rep);
        }

        return vector<string>(result.begin(), result.end());
    }
};
```
