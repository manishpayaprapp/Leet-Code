## Solution

```cpp
class Solution {
public:
    bool circularArrayLoop(vector<int>& nums) {
        int n = nums.size();

        for (int i = 0; i < n; i++) {

            bool positive = nums[i] > 0;
            int curr = i;

            for (int count = 0; count < n; count++) {

                // Direction must remain the same
                if ((nums[curr] > 0) != positive)
                    break;

                int next = ((curr + nums[curr]) % n + n) % n;

                // Single element cycle is not allowed
                if (next == curr)
                    break;

                curr = next;

                // We returned to starting index
                if (curr == i)
                    return true;
            }
        }

        return false;
    }
};
```
