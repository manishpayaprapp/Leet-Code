class Solution {
public:

    int calcNextIdx(vector<int>& nums, int curr) {
        int n = nums.size();
        int seq = nums[curr];

        if (seq > 0) {
            return (curr + seq) % n;
        }

        int mod = seq % n;
        int forward = n + mod;

        return (curr + forward) % n;
    }

    bool circularArrayLoop(vector<int>& nums) {

        int n = nums.size();

        for (int i = 0; i < n; i++) {

            if (nums[i] == 0) {
                continue;
            }

            unordered_set<int> set;
            set.insert(i);

            bool isPos = nums[i] > 0;
            int curr = i;

            while (true) {

                int next = calcNextIdx(nums, curr);

                if (isPos) {

                    // Direction changed
                    if (nums[next] < 0) {
                        break;
                    }

                    if (set.find(next) != set.end()) {

                        // Self loop: invalid
                        if (curr == next) {
                            break;
                        }

                        // Valid cycle
                        return true;
                    }

                    set.insert(next);
                }
                else {

                    // Direction changed
                    if (nums[next] > 0) {
                        break;
                    }

                    if (set.find(next) != set.end()) {

                        // Self loop: invalid
                        if (curr == next) {
                            break;
                        }

                        // Valid cycle
                        return true;
                    }

                    set.insert(next);
                }

                curr = next;
            }

            // Mark this entire path as processed
            curr = i;

            if (isPos) {

                while (nums[curr] > 0) {

                    int next = calcNextIdx(nums, curr);

                    nums[curr] = 0;

                    curr = next;
                }

            }
            else {

                while (nums[curr] < 0) {

                    int next = calcNextIdx(nums, curr);

                    nums[curr] = 0;

                    curr = next;
                }
            }
        }

        return false;
    }
};
