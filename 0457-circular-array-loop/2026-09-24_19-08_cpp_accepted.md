## Solution

```cpp
class Solution {
public:
    int calNext(vector<int>& nums, int curr) {
        int n = nums.size();

        if (nums[curr] > 0) {
            return (curr + nums[curr]) % n;
        }

        return (curr + nums[curr] % n + n) % n;
    }

public:
    bool circularArrayLoop(vector<int>& nums) {

        int n = nums.size();

        for (int i = 0; i < n; i++) {

            unordered_set<int> visited;

            bool isPos = nums[i] > 0;
            int curr = i;

            while (true) {

                int next = calNext(nums, curr);

                // 1. Self loop is invalid
                if (next == curr) {
                    break;
                }

                // 2. Direction changed
                if ((nums[next] > 0) != isPos) {
                    break;
                }

                // 3. We found a cycle
                if (visited.find(next) != visited.end()) {
                    return true;
                }

                visited.insert(curr);

                curr = next;
            }
        }

        return false;
    }
};
```
