## Solution

class Solution {
public:

    int calcNextIdx(vector<int>& nums, int curr) {
        int n = nums.size();
        int seq = nums[curr];

        if (seq > 0) {
            return (curr + seq) % n;
        } 
        else {
            int mod = seq % n;
            int forward = n + mod;
            return (curr + forward) % n;
        }
    }

    bool circularArrayLoop(vector<int>& nums) {

        int n = nums.size();

        for (int i = 0; i < n; i++) {

            if (nums[i] == 0) {
                continue;
            }

            bool isPos = nums[i] > 0;

            int slow = i;
            int fast = i;

            while (true) {

                // Move slow one step
                int nextSlow = calcNextIdx(nums, slow);

                // Direction changed
                if ((nums[nextSlow] > 0) != isPos) {
                    break;
                }

                // Self loop
                if (nextSlow == slow) {
                    break;
                }

                slow = nextSlow;


                // Move fast one step
                int nextFast = calcNextIdx(nums, fast);

                if ((nums[nextFast] > 0) != isPos) {
                    break;
                }

                if (nextFast == fast) {
                    break;
                }

                fast = nextFast;


                // Move fast second step
                nextFast = calcNextIdx(nums, fast);

                if ((nums[nextFast] > 0) != isPos) {
                    break;
                }

                if (nextFast == fast) {
                    break;
                }

                fast = nextFast;


                // Cycle detected
                if (slow == fast) {
                    return true;
                }
            }

            // Mark this path as visited
            int curr = i;

            while (nums[curr] != 0 &&
                   ((nums[curr] > 0) == isPos)) {

                int next = calcNextIdx(nums, curr);

                nums[curr] = 0;

                curr = next;
            }
        }

        return false;
    }
};
