## Solution

```cpp
class Solution {
public:
    vector<int> findRightInterval(vector<vector<int>>& intervals) {
        int len = intervals.size();
        vector<int> res(len, -1);

        // T: O(n^2), S: O(n)
        for (int i = 0; i < len; i++) {
            int minStartIdx = -1;

            for (int j = 0; j < len; j++) {
                if (intervals[j][0] >= intervals[i][1]) {
                    if (minStartIdx == -1) {
                        minStartIdx = j;
                    }

                    if (intervals[j][0] < intervals[minStartIdx][0]) {
                        minStartIdx = j;
                    }
                }
            }

            res[i] = minStartIdx;
        }

        return res;
    }
};
```
