## Solution

```cpp
class Solution {
public:
    vector<int> findRightInterval(vector<vector<int>>& intervals) {
        priority_queue<
        pair<int,int>,
        vector<pair<int,int>>,
        greater<pair<int,int>>
        >minStart;

        priority_queue<
        pair<int,int>,
        vector<pair<int,int>>,
        greater<pair<int,int>>
        >minEnd;

        int len = intervals.size();
        vector<int> res(len,-1);

        for(int i = 0; i < len; i++){
            minStart.push({intervals[i][0],i});
            minEnd.push({intervals[i][1],i});
        }

        while(!minStart.empty() && !minEnd.empty()){
            auto startEle = minStart.top();

            int start = startEle.first;
            int startIdx = startEle.second;

            if(start >= minEnd.top().first){
                res[minEnd.top().second] = startIdx;

                minEnd.pop();
            }
            else{
                minStart.pop();
            }
        }
        return res;
    }
};
```
