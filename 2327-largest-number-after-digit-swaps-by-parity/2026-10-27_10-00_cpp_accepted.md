## Solution

```cpp
class Solution {
public:
    int largestInteger(int num) {
        priority_queue<int> even;
        priority_queue<int> odd;

        string s = to_string(num);

        for (char c : s) {
            int val = c - '0';

            if (val % 2 == 0) {
                even.push(val);
            } else {
                odd.push(val);
            }
        }

        string result = "";

        for (char c : s) {
            int val = c - '0';

            if (val % 2 == 0) {
                result += to_string(even.top());
                even.pop();
            } else {
                result += to_string(odd.top());
                odd.pop();
            }
        }

        return stoi(result);
    }
};
```
