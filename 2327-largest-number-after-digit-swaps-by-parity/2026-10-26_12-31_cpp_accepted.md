## Solution

```cpp
class Solution {
public:
    int largestInteger(int num) {
        string s = to_string(num);
        int n = s.size();

        for (int i = 0; i < n; i++) {
            int maxIdx = i;

            for (int j = i + 1; j < n; j++) {
                if ((s[j] - '0') % 2 == (s[i] - '0') % 2 &&
                    s[j] > s[maxIdx]) {
                    maxIdx = j;
                }
            }

            swap(s[i], s[maxIdx]);
        }

        return stoi(s);
    }
};
```
