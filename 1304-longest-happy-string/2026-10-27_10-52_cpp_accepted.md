class CharacterCount {
public:
    int count;
    char c;

    CharacterCount(int _count, char _c) {
        count = _count;
        c = _c;
    }
};

class Solution {
public:
    string longestDiverseString(int a, int b, int c) {
        // Max Heap: count -> character
        priority_queue<pair<int, char>> maxHeap;

        // Add only characters with positive counts
        if (a > 0) {
            maxHeap.push({a, 'a'});
        }
        if (b > 0) {
            maxHeap.push({b, 'b'});
        }
        if (c > 0) {
            maxHeap.push({c, 'c'});
        }

        string res = "";

        while (!maxHeap.empty()) {
            auto maxEle = maxHeap.top();
            maxHeap.pop();

            int count = maxEle.first;
            char _c = maxEle.second;

            // Avoid three consecutive identical characters
            if (res.length() >= 2 &&
                res[res.length() - 1] == _c &&
                res[res.length() - 2] == _c) {

                if (maxHeap.empty()) {
                    break;
                }

                auto secondMaxEle = maxHeap.top();
                maxHeap.pop();

                int sCount = secondMaxEle.first;
                char sC = secondMaxEle.second;

                res.push_back(sC);
                sCount--;

                if (sCount > 0) {
                    maxHeap.push({sCount, sC});
                }
            } else {
                res.push_back(_c);
                count--;
            }

            if (count > 0) {
                maxHeap.push({count, _c});
            }
        }

        return res;
    }
};
