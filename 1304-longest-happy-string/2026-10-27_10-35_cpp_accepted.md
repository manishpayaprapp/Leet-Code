## Solution

```cpp
class Solution {
public:
    string longestDiverseString(int a, int b, int c) {
        int c_a = 0,
            c_b = 0,
            c_c = 0;

        string res = "";

        while(true){
            if(a > 0 && ((a >= b && a >= c && c_a < 2) || 
            (c_b == 2 && a >= c && b >= a)||
            (c_c == 2 && a >= b && c >= a)))
            {
                res.push_back('a');
                c_a ++;
                c_b = 0;
                c_c = 0;

                a--;
            }
            else if(b > 0 && ((b >= a && b >= c && c_b < 2) || 
            (c_a == 2 && b >= c && a >= b)||
            (c_c == 2 && b >= a && c >= b)))
            {
                res.push_back('b');
                c_a = 0;
                c_b ++;
                c_c = 0;

                b--;
            }
            else if(c > 0 && ((c >= b && c >= a && c_c < 2) || 
            (c_b == 2 && c >= a && b >= c)||
            (c_a == 2 && c >= b && a >= c)))
            {
                res.push_back('c');
                c_a = 0;
                c_b = 0;
                c_c ++;

                c--;
            }
            else{
                break;
            }
        }
        return res;

    }
};
```
