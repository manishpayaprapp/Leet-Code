## Solution

```cpp

class SummaryRanges {
    vector<int> arr;

public:
    SummaryRanges() {}

    void addNum(int value) {
        arr.push_back(value);
    }

    vector<vector<int>> getIntervals() {
        if (arr.empty()) {
            return {};
        }

        sort(arr.begin(), arr.end());

        vector<vector<int>> res;

        int left = arr[0];
        int right = arr[0];

        for (int i = 1; i < arr.size(); i++) {
            if (arr[i] <= right) {
                continue; // Skip duplicates
            }

            if (arr[i] == right + 1) {
                right = arr[i];
            } else {
                res.push_back({left, right});
                left = right = arr[i];
            }
        }

        res.push_back({left, right}); // Last interval

        return res;
    }
};
```
