## Solution

```cpp

class SummaryRanges {
    set<int> st;

public:
    SummaryRanges() {}

    void addNum(int value) {
        st.insert(value); // Automatically sorted and unique
    }

    vector<vector<int>> getIntervals() {
        vector<vector<int>> res;

        if (st.empty()) {
            return res;
        }

        auto it = st.begin();
        int left = *it;
        int right = *it;
        ++it;

        for (; it != st.end(); ++it) {
            int num = *it;

            if (num == right + 1) {
                right = num;
            } else {
                res.push_back({left, right});
                left = right = num;
            }
        }

        res.push_back({left, right});

        return res;
    }
};
```
