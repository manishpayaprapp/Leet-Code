## Solution

```cpp
class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        int sum = 0;
        int size = numeric_limits<int>::max();

        int i = 0;
        int j = 0;

        while(j < nums.size()){
            sum = sum + nums[j];

            while(sum >= target){
                size = min(size,j - i +1);

                sum = sum - nums[i];
                i++;
            }
            j++;
        }
        return size  == numeric_limits<int>::max() ? 0 : size;
    }
};
```
