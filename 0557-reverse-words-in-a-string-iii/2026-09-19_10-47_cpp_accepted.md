## Solution

```cpp
class Solution {
public:
    string reverseWords(string s) {
        for (int i = 0, j = 0; i <= s.size(); i++) {
            if (i == s.size() || s[i] == ' ') {
                for (int l = j, r = i - 1; l < r; l++, r--)
                    swap(s[l], s[r]);
                j = i + 1;
            }
        }
        return s;
    }
};
```
