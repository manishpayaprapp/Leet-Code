## Solution

```cpp
class Solution {
public:
    string reverseWords(string s) {
       int start = 0, end = 0;
        int n = s.size();

        while (start < n) {
            while (end < n && s[end] != ' ')
                end++;

            reverse(s.begin() + start, s.begin() + end);

            end++;
            start = end;
        }
        return s;
    }
};
```
