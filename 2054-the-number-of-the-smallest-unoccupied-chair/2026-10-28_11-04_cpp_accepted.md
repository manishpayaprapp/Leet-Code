## Solution

```cpp
class Solution {
public:
    int smallestChair(vector<vector<int>>& times, int targetFriend) {
        int targetFriendArrivalTime = times[targetFriend][0];

        sort(times.begin(), times.end());

        priority_queue<int, vector<int>, greater<int>> availableChairs;

        // Min heap: {endTime, chairIndex}
        priority_queue<pair<int, int>,
                       vector<pair<int, int>>,
                       greater<pair<int, int>>> occupiedChairs;

        // Add all chairs initially
        for (int i = 0; i < times.size(); i++) {
            availableChairs.push(i);
        }

        // T: O(n log n), S: O(n)
        for (int i = 0; i < times.size(); i++) {

            int aT = times[i][0];

            // Release chairs of friends who have already left
            while (!occupiedChairs.empty()) {
                if (aT >= occupiedChairs.top().first) {
                    int chairIndex = occupiedChairs.top().second;
                    occupiedChairs.pop();

                    availableChairs.push(chairIndex);
                } else {
                    break;
                }
            }

            // Assign the smallest available chair
            int c = availableChairs.top();
            availableChairs.pop();

            // Check if this is the target friend
            if (targetFriendArrivalTime == aT) {
                return c;
            }

            // Occupy the chair until the friend's departure
            occupiedChairs.push({times[i][1], c});
        }

        return -1;
    }
};
```
